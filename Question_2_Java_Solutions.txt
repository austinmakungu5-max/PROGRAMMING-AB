QUESTION 2 – SOLUTIONS
Java Mattress Sales Application

The solution below uses Java Swing for the GUI and JUnit 5 for the unit tests.
Create the Java files in the same package (or remove the package declaration if
your project does not use packages).

============================================================
1. IProductCalculations.java
============================================================

public interface IProductCalculations {
    double GetVat();
    double GetCommission();
    double GetTotal();
}


============================================================
2. ProductCalculations.java
============================================================

public class ProductCalculations implements IProductCalculations {

    private double price;

    public ProductCalculations(double price) {
        this.price = price;
    }

    @Override
    public double GetVat() {
        return price * 0.15;
    }

    @Override
    public double GetCommission() {
        return price * 0.05;
    }

    @Override
    public double GetTotal() {
        return price + GetVat() + GetCommission();
    }
}

3. MattressSalesApp.java
============================================================

import javax.swing.*;
import java.awt.*;
import java.io.FileWriter;
import java.io.IOException;
import java.io.PrintWriter;

public class MattressSalesApp extends JFrame {

    private JTextField txtCustomerName;
    private JComboBox<String> cmbProduct;
    private JTextField txtPrice;
    private JButton btnProcess;

    public MattressSalesApp() {
        setTitle("Product App");
        setSize(500, 300);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        // Create input fields
        txtCustomerName = new JTextField(20);

        String[] products = {
            "Select Product",
            "Cozy Nights Monarch",
            "Restonic Barbados",
            "Sealy Phuket"
        };
        cmbProduct = new JComboBox<>(products);

        txtPrice = new JTextField(20);
        btnProcess = new JButton("PROCESS");

        // Create labels and arrange the form
        JPanel panel = new JPanel(new GridLayout(4, 2, 10, 15));
        panel.setBorder(BorderFactory.createEmptyBorder(25, 25, 25, 25));

        panel.add(new JLabel("Customer Name:"));
        panel.add(txtCustomerName);

        panel.add(new JLabel("Select Product:"));
        panel.add(cmbProduct);

        panel.add(new JLabel("Enter Price:"));
        panel.add(txtPrice);

        panel.add(new JLabel(""));
        panel.add(btnProcess);

        add(panel);

        // Button calls the process method
        btnProcess.addActionListener(e -> processSale());

        // Create menu
        JMenuBar menuBar = new JMenuBar();

        JMenu fileMenu = new JMenu("File");
        JMenuItem exitItem = new JMenuItem("Exit");
        exitItem.addActionListener(e -> System.exit(0));
        fileMenu.add(exitItem);

        JMenu toolsMenu = new JMenu("Tools");

        JMenuItem processItem = new JMenuItem("Process");
        processItem.addActionListener(e -> processSale());

        JMenuItem clearItem = new JMenuItem("Clear");
        clearItem.addActionListener(e -> clearForm());

        toolsMenu.add(processItem);
        toolsMenu.add(clearItem);

        menuBar.add(fileMenu);
        menuBar.add(toolsMenu);
        setJMenuBar(menuBar);
    }

    // This method processes the sale
    private void processSale() {

        String customerName = txtCustomerName.getText().trim();
        String productName = (String) cmbProduct.getSelectedItem();
        String priceText = txtPrice.getText().trim();

        // Validate the input
        if (customerName.isEmpty()) {
            JOptionPane.showMessageDialog(this,
                    "Please enter the customer name.");
            return;
        }

        if (cmbProduct.getSelectedIndex() == 0) {
            JOptionPane.showMessageDialog(this,
                    "Please select a product.");
            return;
        }

        double price;

        try {
            price = Double.parseDouble(priceText);

            if (price <= 0) {
                JOptionPane.showMessageDialog(this,
                        "Please enter a price greater than zero.");
                return;
            }

        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this,
                    "Please enter a valid numeric price.");
            return;
        }

        // Create calculation object
        ProductCalculations calculation =
                new ProductCalculations(price);

        double vat = calculation.GetVat();
        double commission = calculation.GetCommission();
        double total = calculation.GetTotal();

        // Display the results
        String message =
                "Customer Name: " + customerName +
                "\nProduct: " + productName +
                "\nVAT: R " + vat +
                "\nCOMMISSION: R " + commission +
                "\nTOTAL: R " + total;

        JOptionPane.showMessageDialog(this, message);

        // Save the information to data.txt
        saveToFile(customerName, productName, vat, commission, total);
    }

    // Save sale information to a text file
    private void saveToFile(String customerName, String productName,
                            double vat, double commission, double total) {

        try (PrintWriter writer =
                     new PrintWriter(new FileWriter("data.txt", true))) {

            writer.println("DATA LOG");
            writer.println("****************************************");
            writer.println("Customer Name: " + customerName);
            writer.println("Product: " + productName);
            writer.println("VAT: R " + vat);
            writer.println("COMMISSION: R " + commission);
            writer.println("TOTAL: R " + total);
            writer.println("****************************************");
            writer.println();

        } catch (IOException ex) {
            JOptionPane.showMessageDialog(this,
                    "Error saving data to file: " + ex.getMessage());
        }
    }

    // Clear all text boxes and reset the combo box
    private void clearForm() {
        txtCustomerName.setText("");
        txtPrice.setText("");
        cmbProduct.setSelectedIndex(0);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            new MattressSalesApp().setVisible(true);
        });
    }
}

4. UNIT TESTS – JUNIT 5
============================================================

Create a test package in the project and add the following test class.

ProductCalculationsTest.java

import static org.junit.jupiter.api.Assertions.assertEquals;
import org.junit.jupiter.api.Test;

public class ProductCalculationsTest {

    public void GetVat_WithValidData_ReturnsCorrectValue() {

                new ProductCalculations(3500);

        assertEquals(525.0, calculation.GetVat(), 0.001);
    }

    @Test
    public void GetCommission_WithValidData_ReturnsCorrectValue() {
        ProductCalculations calculation =
                new ProductCalculations(3500);

        assertEquals(175.0, calculation.GetCommission(), 0.001);
    }

    @Test
    public void GetTotal_WithValidData_ReturnsCorrectValue() {
        ProductCalculations calculation =
                new ProductCalculations(3500);

        assertEquals(4200.0, calculation.GetTotal(), 0.001);
    }
}

5. CALCULATION EXAMPLE
============================================================

Given:
Product price = R 3500

VAT = 15% of product price
VAT = 3500 x 15 / 100
VAT = R 525

Commission = 5% of product price
Commission = 3500 x 5 / 100
Commission = R 175

Total Sale Amount = Price + VAT + Commission
Total = 3500 + 525 + 175
Total = R 4200

7. HOW TO RUN THE APPLICATION
============================================================

1. Open your Java IDE (for example, NetBeans).
2. Create a Java application project.
3. Create the interface named IProductCalculations.
4. Create the class named ProductCalculations.
5. Create the JFrame class named MattressSalesApp.
6. Copy each section of code into its matching Java file.
7. Run MattressSalesApp.
8. Enter the customer name, select a mattress, and enter its price.
9. Click PROCESS. The results will appear and be saved in data.txt.
10. Add JUnit 5 to the project before running ProductCalculationsTest.

NOTE:
The data.txt file is created in the application's working directory.
The application appends each new sale to the file instead of deleting
previous sales.
